a.

SELECT j.Job_Title, COUNT(e.employee_id)
FROM employees e, jobs j, departments d
WHERE e.department_id = d.department_id AND e.job_ID = j.job_ID AND d.department_name = 'Accounting' OR 'Shipping'
GROUP BY Job_Title
ORDER BY COUNT(e.employee_id) DESC;

b.

SELECT j.Job_Title, COUNT(e.employee_id)
FROM employees e JOIN jobs j JOIN departments d
	ON e.department_id = d.department_id AND e.job_ID = j.job_ID 
WHERE d.department_name = 'Accounting' OR d.department_name = 'Shipping'
GROUP BY Job_Title
ORDER BY COUNT(e.employee_id) DESC;

FROM employees e, employees m
WHERE m.employee_id = e.manager_id AND m.Salary > 15000;

b.

SELECT e.last_name EMPLOYEE, m.last_name MANAGER, m.salary 'MANAGER Salary'
FROM employees e JOIN employees m
	ON m.employee_id = e.manager_id
