DROP Procedure PView2;
Delimiter //
CREATE PROCEDURE PView2(IN ONumb varchar(6), OUT TotalP decimal(8.2)) 
BEGIN
    Select SUM(Profit) INTO TotalP
	From Profits
	Where OrderNO = ONumb;

Ex5. Change the department number where an employee works.

Drop Procedure ChangeWhereEMPWorks;
Delimiter //
CREATE PROCEDURE ChangeWhereEMPWorks(IN IdEmp INT, IN DepId INT)
BEGIN
    DECLARE Messa VARCHAR(20) default null;
	SET @EFlag = Null;
	SET @EFlag = (Select department_id 
	              From employees 
				  Where employee_id = IdEmp);
	IF @EFlag IS NULL THEN 
	    SET Messa = '*** Id Not Found ***';
	  ELSE 
	    UPDATE EMPLOYEES
		SET department_id = DepId 
		Where employee_id = IdEmp;
		SET Messa = '** Update Done ***';
	END IF;
	
	Select Messa as Message;
END//

DROP PROCEDURE division;
DELIMITER //

CREATE PROCEDURE division(IN num1 INT, IN num2 INT)
BEGIN
	DECLARE result1 INT default 0;
	IF num2 = 0 THEN
		SELECT result1 AS '**CANNOT DIVIDE BY 0**';
	ELSE 
		SET result1 = num1 / num2;
	SELECT result1 as 'div';
	END IF;
END //

DELIMITER ;
CALL division(10,0);

DROP PROCEDURE mul;

DELIMITER //

CREATE PROCEDURE mul(IN n1 INT, IN n2 INT)
BEGIN
	DECLARE r INT default 0;
	SET r = n1*n2;
	IF n1 = 0 OR n2 = 0 THEN
		SELECT r AS '**0 rlly?**';
	ELSE
		SELECT r AS multiplication;
	END IF;
END //

DELIMITER ;
CALL mul(5,5);
